# 🎮 霓虹空战 - 资产管理系统设计方案

> 版本: v1.0  
> 日期: 2025-12-26  
> 目标: 将硬编码资源分离为可维护、可扩展的资产系统

---

## 📊 当前状态分析

### 现有资源分布

| 资源类型 | 当前实现 | 代码量 | 问题 |
|---------|---------|--------|------|
| 机体外观 | Python代码绘制 | ~60个文件, 数万行 | 修改需改代码 |
| 子弹特效 | Python代码绘制 | ~60个文件 | 无法热更新 |
| 音乐/音效 | 程序化合成 | ~2000行 | 启动时生成慢 |
| 机体数据 | `config.py` 硬编码 | ~800行 | 混合业务逻辑 |
| 涂装配置 | JSON + Python混合 | 分散 | 不统一 |
| UI布局 | `main.py` 硬编码 | ~500行 | 难以调整 |

### 核心问题

1. **代码膨胀**: `sprites.py` 已达 **20,864 行**
2. **耦合严重**: 数据、逻辑、渲染混在一起
3. **无法热更新**: 任何视觉修改都需要改代码
4. **协作困难**: 美术无法独立修改资源
5. **加载缓慢**: 音频需要运行时合成

### 1. 资产管理器 (`src/assets/asset_manager.py`)

```python
"""
统一资产管理系统

功能：
- 懒加载：按需加载资源
- 缓存：避免重复加载
- 热重载：开发模式支持实时更新
- 回退：缺失资源使用默认替代
"""
import os
import json
import pygame
from typing import Dict, Any, Optional
from pathlib import Path


class AssetManager:
    """单例资产管理器"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
        self._initialized = True
        
        # 路径配置
        self.base_path = Path(__file__).parent.parent.parent
        self.assets_path = self.base_path / "assets"
        self.data_path = self.base_path / "data"
        
        # 缓存
        self._sprite_cache: Dict[str, pygame.Surface] = {}
        self._sound_cache: Dict[str, pygame.mixer.Sound] = {}
        self._data_cache: Dict[str, dict] = {}
        self._font_cache: Dict[tuple, pygame.font.Font] = {}
        
        # 配置
        self.dev_mode = os.environ.get("DEV_MODE", "0") == "1"
        self._file_watchers = {}
        
        # 预加载基础资源
        self._load_color_palette()
    
    # ==================== 精灵加载 ====================
    
    def get_sprite(self, path: str, scale: float = 1.0) -> pygame.Surface:
        """获取精灵图像"""
        cache_key = f"{path}@{scale}"
        
        if cache_key in self._sprite_cache:
            return self._sprite_cache[cache_key]
        
        full_path = self.assets_path / path
        
        try:
            surface = pygame.image.load(str(full_path)).convert_alpha()
            if scale != 1.0:
                new_size = (int(surface.get_width() * scale),
                           int(surface.get_height() * scale))
                surface = pygame.transform.scale(surface, new_size)
            self._sprite_cache[cache_key] = surface
            return surface
        except FileNotFoundError:
            return self._get_fallback_sprite(path)
    
    def get_sprite_sheet(self, path: str, frame_size: tuple, 
                         frame_count: int) -> list[pygame.Surface]:
        """加载精灵表并切割为帧列表"""
        sheet = self.get_sprite(path)
        frames = []
        
        for i in range(frame_count):
            frame = pygame.Surface(frame_size, pygame.SRCALPHA)
            frame.blit(sheet, (0, 0), 
                      (i * frame_size[0], 0, frame_size[0], frame_size[1]))
            frames.append(frame)
        
        return frames
    
    def _get_fallback_sprite(self, path: str) -> pygame.Surface:
        """获取回退精灵（缺失资源时使用）"""
        # 创建一个带X的红色方块表示缺失
        size = (64, 64)
        surf = pygame.Surface(size, pygame.SRCALPHA)
        pygame.draw.rect(surf, (255, 0, 0), (0, 0, *size), 2)
        pygame.draw.line(surf, (255, 0, 0), (0, 0), size, 2)
        pygame.draw.line(surf, (255, 0, 0), (size[0], 0), (0, size[1]), 2)
        print(f"[AssetManager] Missing sprite: {path}")
        return surf
    
    # ==================== 数据加载 ====================
    
    def get_data(self, path: str) -> dict:
        """获取JSON数据配置"""
        if path in self._data_cache and not self.dev_mode:
            return self._data_cache[path]
        
        full_path = self.data_path / path
        
        try:
            with open(full_path, 'r', encoding='utf-8') as f:
                data = json.load(f)
            self._data_cache[path] = data
            return data
        except FileNotFoundError:
            print(f"[AssetManager] Missing data: {path}")
            return {}
    
    def get_plane_config(self, plane_id: str) -> dict:
        """获取机体配置"""
        return self.get_data(f"planes/{plane_id}.json")
    
    def get_skin_config(self, plane_id: str) -> dict:
        """获取涂装配置"""
        return self.get_data(f"skins/{plane_id}_skins.json")
    
    def get_all_planes(self) -> list[str]:
        """获取所有机体ID列表"""
        planes_dir = self.data_path / "planes"
        return [f.stem for f in planes_dir.glob("*.json") 
                if not f.name.startswith("_")]
    
    # ==================== 音频加载 ====================
    
    def get_sound(self, name: str) -> Optional[pygame.mixer.Sound]:
        """获取音效"""
        if name in self._sound_cache:
            return self._sound_cache[name]
        
        manifest = self.get_data("audio/sfx_manifest.json")
        sound_info = manifest.get("sounds", {}).get(name)
        
        if not sound_info:
            print(f"[AssetManager] Unknown sound: {name}")
            return None
        
        base_path = manifest.get("base_path", "assets/audio/sfx")
        file_path = self.base_path / base_path / sound_info["file"]
        
        try:
            sound = pygame.mixer.Sound(str(file_path))
            sound.set_volume(sound_info.get("volume", 0.5))
            self._sound_cache[name] = sound
            return sound
        except:
            print(f"[AssetManager] Failed to load sound: {name}")
            return None
    
    # ==================== 字体加载 ====================
    
    def get_font(self, size: int, bold: bool = False) -> pygame.font.Font:
        """获取字体"""
        cache_key = (size, bold)
        
        if cache_key in self._font_cache:
            return self._font_cache[cache_key]
        
        # 尝试加载自定义字体
        font_path = self.assets_path / "fonts" / "main_font.ttf"
        
        try:
            if font_path.exists():
                font = pygame.font.Font(str(font_path), size)
            else:
                # 回退到系统字体
                font = pygame.font.SysFont(
                    ["microsoftyahei", "simhei", "arial"], 
                    size, bold=bold
                )
            self._font_cache[cache_key] = font
            return font
        except:
            return pygame.font.Font(None, size)
    
    # ==================== 颜色系统 ====================
    
    def _load_color_palette(self):
        """加载颜色配置"""
        self.colors = {}
        try:
            data = self.get_data("ui/colors.json")
            self._parse_colors(data.get("palette", {}))
        except:
            self._use_default_colors()
    
    def _parse_colors(self, palette: dict, prefix: str = ""):
        """递归解析颜色配置"""
        for key, value in palette.items():
            full_key = f"{prefix}_{key}" if prefix else key
            if isinstance(value, dict):
                self._parse_colors(value, full_key)
            elif isinstance(value, str) and value.startswith("#"):
                self.colors[full_key] = self._hex_to_rgb(value)
            elif isinstance(value, (list, tuple)):
                self.colors[full_key] = tuple(value)
    
    def _hex_to_rgb(self, hex_color: str) -> tuple:
        """将十六进制颜色转换为RGB"""
        hex_color = hex_color.lstrip("#")
        return tuple(int(hex_color[i:i+2], 16) for i in (0, 2, 4))
    
    def _use_default_colors(self):
        """使用默认颜色（回退）"""
        self.colors = {
            "primary_cyan": (0, 255, 255),
            "accent_amber": (255, 215, 0),
            "alert_red": (255, 51, 51),
        }
    
    def get_color(self, name: str) -> tuple:
        """获取颜色"""
        return self.colors.get(name, (255, 255, 255))
    
    # ==================== 缓存管理 ====================
    
    def clear_cache(self, cache_type: str = "all"):
        """清理缓存"""
        if cache_type in ("all", "sprites"):
            self._sprite_cache.clear()
        if cache_type in ("all", "sounds"):
            self._sound_cache.clear()
        if cache_type in ("all", "data"):
            self._data_cache.clear()
        if cache_type in ("all", "fonts"):
            self._font_cache.clear()
    
    def preload_essential(self):
        """预加载核心资源"""
        # 预加载常用精灵
        essential_sprites = [
            "sprites/ui/cursor.png",
            "sprites/ui/health_bar.png",
        ]
        for path in essential_sprites:
            self.get_sprite(path)
        
        # 预加载常用音效
        essential_sounds = ["shoot", "hit", "explosion", "select"]
        for name in essential_sounds:
            self.get_sound(name)
    
    def get_memory_usage(self) -> dict:
        """获取内存使用情况"""
        return {
            "sprites": len(self._sprite_cache),
            "sounds": len(self._sound_cache),
            "data": len(self._data_cache),
            "fonts": len(self._font_cache),
        }


# 全局单例
asset_manager = AssetManager()
```

def generate_manifest(sprite_dir: str):
    """生成精灵清单文件"""
    import json
    
    manifest = {
        "version": "1.0",
        "sprites": {}
    }
    
    for root, dirs, files in os.walk(sprite_dir):
        for file in files:
            if file.endswith(".png"):
                rel_path = os.path.relpath(
                    os.path.join(root, file), 
                    sprite_dir
                )
                # 获取图像尺寸
                img = pygame.image.load(os.path.join(root, file))
                manifest["sprites"][rel_path] = {
                    "width": img.get_width(),
                    "height": img.get_height()
                }
    
    manifest_path = os.path.join(sprite_dir, "manifest.json")
    with open(manifest_path, "w", encoding="utf-8") as f:
        json.dump(manifest, f, indent=2, ensure_ascii=False)
    print(f"Generated: {manifest_path}")

### 3. 音频预构建工具 (`tools/audio_prebuild.py`)

```python
"""
音频预构建工具

将程序化生成的音频预先生成并保存到assets目录
"""
import os
import sys
import shutil

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from utils.audio import AudioSynthesizer


def prebuild_audio(output_dir: str = "assets/audio"):
    """预构建所有音频"""
    sfx_dir = os.path.join(output_dir, "sfx")
    bgm_dir = os.path.join(output_dir, "bgm")
    
    os.makedirs(sfx_dir, exist_ok=True)
    os.makedirs(bgm_dir, exist_ok=True)
    
    synth = AudioSynthesizer()
    paths = synth.generate_all()
    
    # 复制生成的文件到assets目录
    for name, path in paths.items():
        if not path or not os.path.exists(path):
            continue
        
        if name.startswith("bgm"):
            dest = os.path.join(bgm_dir, os.path.basename(path))
        else:
            dest = os.path.join(sfx_dir, os.path.basename(path))
        
        shutil.copy2(path, dest)
        print(f"Copied: {dest}")
    
    print(f"\n音频预构建完成！共 {len(paths)} 个文件")


if __name__ == "__main__":
    prebuild_audio()
```
