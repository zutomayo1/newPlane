## 概述

添加一架新机体需要修改/创建以下文件：

| 文件 | 用途 |
|------|------|
| `config.py` | 定义机体基础属性（HP、速度、伤害等） |
| `utils/planes/skins_xxx.py` | 机体外形渲染（新建文件） |
| `utils/planes/base.py` | 注册机体渲染入口 |
| `utils/planes/__init__.py` | 导出新模块 |
| `sprites.py` | 固有能力和三个大招实现 |
| `utils/bullets/xxx_bullets.py` | 子弹特效渲染（新建文件） |
| `utils/bullets/__init__.py` | 更新注释 |
| `customization.py` | 专属涂装和子弹涂装定义 |

```python
        return None
```

### 绘图技巧

1. **Surface 大小**：始终使用 120x120 像素
2. **中心点**：(60, 60) 是机体中心
3. **透明度**：使用 `pygame.SRCALPHA` 创建带透明度的 Surface
4. **动态效果**：
   - `t` 参数：时间（秒），用于旋转动画
   - `pulse` 参数：0-1 的脉冲值，用于呼吸效果
5. **颜色透明度**：使用 `(*color, alpha)` 格式

## 步骤六：创建子弹效果模块

### 创建文件
`utils/bullets/xxx_bullets.py`

### 文件模板

```python
# -*- coding: utf-8 -*-
"""
XXX 战机子弹涂装效果渲染模块

包含以下子弹效果：
- xxx_basic: 基础子弹
- xxx_effect1: 特效1
- xxx_effect2: 特效2
"""
import pygame
import math


def render_xxx_bullet(surface, effects, color, center_x, center_y, size, x, y):
    """
    渲染 XXX 战机的子弹效果
    
    Args:
        surface: pygame绘图表面
        effects: 效果列表 (从子弹涂装配置的 "effects" 字段)
        color: 主题颜色
        center_x, center_y: 中心坐标
        size: 子弹大小
        x, y: 左上角坐标
    
    Returns:
        bool: 如果渲染了效果返回 True，否则 False
    """
    t = pygame.time.get_ticks() / 1000.0
    
    if "xxx_basic" in effects:
        # 基础子弹效果
        pygame.draw.circle(surface, color, (center_x, center_y), size // 2)
        pygame.draw.circle(surface, (255, 255, 255), (center_x, center_y), size // 4)
        return True
    
    elif "xxx_effect1" in effects:
        # 特效1
        # 旋转星形
        for i in range(6):
            angle = (i * 60 + t * 100) * math.pi / 180
            r = size // 2
            ex = center_x + int(math.cos(angle) * r)
            ey = center_y + int(math.sin(angle) * r)
            pygame.draw.line(surface, color, (center_x, center_y), (ex, ey), 2)
        return True
    
    elif "xxx_effect2" in effects:
        # 特效2
        # 环形光晕
        for ring in range(3):
            ring_r = size // 2 - ring * (size // 8)
            alpha = 200 - ring * 50
            ring_surf = pygame.Surface((size*2, size*2), pygame.SRCALPHA)
            pygame.draw.circle(ring_surf, (*color, alpha), (size, size), ring_r, 2)
            surface.blit(ring_surf, (x - size//2, y - size//2))
        return True
    
    return False
```

```python
# -*- coding: utf-8 -*-
"""星海巡航者涂装"""
import pygame
import math

STARSEA_STYLES = [
    "starsea_nebula",
    "starsea_aurora",
    "starsea_void",
    "starsea_ex",
    "starsea_ex2",
]

def is_starsea_style(model_style):
    return model_style in STARSEA_STYLES

def render_starsea_skin(s, c, model_style, t, pid, static):
    if not is_starsea_style(model_style):
        return None
    pulse = 0 if static else abs(math.sin(t * 3))
    _render_starsea_base(s, t, pulse)
    return s

def _render_starsea_base(s, t, pulse):
    # 星海主题渲染
    blue = (100, 180, 255)
    gold = (255, 220, 150)
    
    # 星图背景
    for i in range(12):
        angle = (i * 30 + t * 20) * 0.01745
        r = 45 + int(5 * math.sin(t * 2 + i))
        x = 60 + math.cos(angle) * r
        y = 60 + math.sin(angle) * r
        pygame.draw.circle(s, gold, (int(x), int(y)), 2 + int(pulse))
    
    # 飞船主体
    pygame.draw.polygon(s, blue, [(60, 25), (35, 85), (85, 85)])
    pygame.draw.polygon(s, gold, [(60, 25), (35, 85), (85, 85)], 2)
    
    # 核心
    pygame.draw.circle(s, (255, 255, 255), (60, 55), 12)
    pygame.draw.circle(s, blue, (60, 55), 8)
```
